kernel跟踪机制

kernel中有很多调试跟踪机制

# ftrace

ftrace 是内核自带的强大跟踪工具，可以用来追踪内核函数调用、调度行为、中断处理等。它通过 /sys/kernel/tracing 目录提供用户接口，可以配置、启用和读取各种跟踪事件

# tracefs文件系统

tracefs 是用来接入内核的 ftrace 功能的文件系统，tracefs 被挂载在 /sys/kernel/tracing 或者 /sys/kernel/debug/tracing（取决于内核版本和配置）。

# tracer

# 查看当前的追踪器
root@ubuntu:/sys/kernel/tracing# cat current_tracer 
nop

# 查看可用的追踪器
root@ubuntu:/sys/kernel/tracing# cat available_tracers 
blk nop

# 开启追踪器，以添加到可用打开，需要编译kernel配置
# CONFIG_BOOTTIME_TRACING is not set
# CONFIG_FUNCTION_TRACER is not set
# CONFIG_STACK_TRACER is not set
# CONFIG_IRQSOFF_TRACER is not set
# CONFIG_SCHED_TRACER is not set
# CONFIG_HWLAT_TRACER is not set
# CONFIG_MMIOTRACE is not set
# CONFIG_FTRACE_SYSCALLS is not set
# CONFIG_TRACER_SNAPSHOT is not set

区分混淆

特性 Event Available Tracers
作用 跟踪内核中特定的事件或行为 提供不同类型的通用跟踪器以跟踪内核的运行行为
用途 聚焦于特定事件的细粒度追踪，例如调度、系统调用 提供整体性或特定类型的内核行为追踪，例如函数调用、调度
配置方式 通过启用某个事件 (echo 1 > event/enable) 通过设置 current_tracer 选择某个跟踪器
查看方式 /sys/kernel/debug/tracing/available_events /sys/kernel/debug/tracing/available_tracers

# event

当跟踪器是nop的时候，会关注追踪events事件

turn on & off

# 启动跟踪子系统，一般情况下是默认开启的
CONFIG_TRACING=y 
CONFIG_FTRACE=y

# 打开跟踪的节点，一般情况下也是默认开启的
echo 1 > /sys/kernel/tracing/tracing_on

# 设置当前的跟踪记录器，可能的值包括：function, sched, nop, syscalls
echo function > /sys/kernel/tracing/current_tracer 

# 读取跟踪数据
cat /sys/kernel/tracing/trace

# 实时读取
cat /sys/kernel/tracing/trace_pipe

# 代码中添加ftrace

使用方式与 printk() 类似，用于向 ftrace 跟踪缓冲区输出跟踪信息。配置CONFIG_TRACING选项后就会定义trace_printk()宏（见include/linux/kernel.h）; 就是在内核代码中嵌入trace_printk()语句，这样在用ftrace的function_graph跟踪器对内核进行监控时就能在trace文件中看到trace_printk()输出的信息。

/*                                                     
* ftrace_demo.c 
*/                                                    
#include <linux/init.h> 
#include <linux/module.h> 
#include <linux/kernel.h> 
 
MODULE_LICENSE("GPL"); 
 
static int ftrace_demo_init(void) 
{ 
    trace_printk("Can not see this in trace unless loaded for the second time\n"); 
    return 0; 
} 
 
static void ftrace_demo_exit(void) 
{ 
    trace_printk("Module unloading\n"); 
} 
 
module_init(ftrace_demo_init); 
module_exit(ftrace_demo_exit);

对模块ftrace_demo进行跟踪：

[root@linux tracing]# echo 1 > /proc/sys/kernel/ftrace_enabled 
[root@linux tracing]# echo function_graph > current_tracer 
# 事先加载模块 ftrace_demo (加载后才能在写文件 set_ftrace_filter 时找到该模块)
 
[root@linux tracing]# echo ':mod:ftrace_demo' > set_ftrace_filter 
[root@linux tracing]# cat set_ftrace_filter 
ftrace_demo_init 
ftrace_demo_exit 
 
# 将模块 ftrace_demo 卸载
 
[root@linux tracing]# echo 1 > tracing_enabled 
 
# 重新进行模块 ftrace_demo 的加载与卸载操作
 
[root@linux tracing]# cat trace

    public void startTransitionBoostDuration(long duration, boolean enter, String callerPkg,

/**
     * Called when the transition is ready to be started, and all leashes have been set up.

    void goodToGo(@WindowManager.TransitionOldType int transit) {
        ProtoLog.d(WM_DEBUG_REMOTE_ANIMATIONS, "goodToGo()");
        if (mCanceled) {
            ProtoLog.d(WM_DEBUG_REMOTE_ANIMATIONS,
                    "goodToGo(): Animation canceled already");
            onAnimationFinished();
            invokeAnimationCancelled("already_cancelled");
            return;
        }

        // Scale the timeout with the animator scale the controlling app is using.
        mHandler.postDelayed(mTimeoutRunnable,
                (long) (TIMEOUT_MS * mService.getCurrentAnimatorScale()));
        mFinishedCallback = new FinishedCallback(this);

        // Create the app targets
        final RemoteAnimationTarget[] appTargets = createAppAnimations();
        if (appTargets.length == 0 && !AppTransition.isKeyguardOccludeTransitOld(transit)) {
            // Keyguard occlude transition can be executed before the occluding activity becomes
            // visible. Even in this case, KeyguardService expects to receive binder call, so we
            // don't cancel remote animation.
            ProtoLog.d(WM_DEBUG_REMOTE_ANIMATIONS,
                    "goodToGo(): No apps to animate, mPendingAnimations=%d",
                    mPendingAnimations.size());
            onAnimationFinished();
            invokeAnimationCancelled("no_app_targets");
            return;
        }

        if (mOnRemoteAnimationReady != null) {
            mOnRemoteAnimationReady.run();
            mOnRemoteAnimationReady = null;
        }

        // Create the remote wallpaper animation targets (if any)
        final RemoteAnimationTarget[] wallpaperTargets = createWallpaperAnimations();

        // Create the remote non app animation targets (if any)
        final RemoteAnimationTarget[] nonAppTargets = createNonAppWindowAnimations(transit);

        mService.mAnimator.addAfterPrepareSurfacesRunnable(() -> {
            try {
                linkToDeathOfRunner();
                ProtoLog.d(WM_DEBUG_REMOTE_ANIMATIONS, "goodToGo(): onAnimationStart,"
                                + " transit=%s, apps=%d, wallpapers=%d, nonApps=%d",
                        AppTransition.appTransitionOldToString(transit), appTargets.length,
                        wallpaperTargets.length, nonAppTargets.length);
                // SDD:add for CIQNHS-9812 by chengqi.ma start
                if (android.os.Build.IS_DEBUG_ENABLE) {
                    Slog.d(TAG, "goodToGo(): onAnimationStart transit = " + AppTransition.appTransitionOldToString(transit) +
                           ", apps = " + appTargets.length +
                           ", wallpapers = " + wallpaperTargets.length +
                           ", nonApps = " + nonAppTargets.length);
                }
                // SDD:add for CIQNHS-9812 by chengqi.ma end
                
                //进行场景6操作
                //T-HUB Core[SPD]:added for animation boost by song.tang 20230619 start
                ITranAnimationBoost.Instance().startRemoteAnimationBoost(transit);
                //T-HUB Core[SPD]:added for animation boost by song.tang 20230619 end
                mRemoteAnimationAdapter.getRunner().onAnimationStart(transit, appTargets,
                        wallpaperTargets, nonAppTargets, mFinishedCallback);
            } catch (RemoteException e) {
                Slog.e(TAG, "Failed to start remote animation", e);
                onAnimationFinished();
            }
            if (ProtoLogImpl.isEnabled(WM_DEBUG_REMOTE_ANIMATIONS)) {
                ProtoLog.d(WM_DEBUG_REMOTE_ANIMATIONS, "startAnimation(): Notify animation start:");
                writeStartDebugStatement();
            }
        });
        setRunningRemoteAnimation(true);

/** Asks the transition player (shell) to start a created but not yet started transition. */
    @NonNull
    Transition requestStartTransition(@NonNull Transition transition, @Nullable Task startTask,
            @Nullable RemoteTransition remoteTransition,
            @Nullable TransitionRequestInfo.DisplayChange displayChange) {
        if (mIsWaitingForDisplayEnabled) {
            ProtoLog.v(ProtoLogGroup.WM_DEBUG_WINDOW_TRANSITIONS, "Disabling player for transition"
                    + " #%d because display isn't enabled yet", transition.getSyncId());

            mAtm.mH.post(() -> mAtm.mWindowOrganizerController.startTransition(

            return transition;
        }
        if (mTransitionPlayer == null || transition.isAborted()) {
            // Apparently, some tests will kill(and restart) systemui, so there is a chance that
            // the player might be transiently null.
            if (transition.isCollecting()) {
                transition.abort();
            }
            return transition;
        }
        //这里就是我们的代码被调用的地方startTransitionBoost；因此这里的hook点就是应用启动的hook点、

        try {
            ProtoLog.v(ProtoLogGroup.WM_DEBUG_WINDOW_TRANSITIONS,

            ActivityManager.RunningTaskInfo info = null;

# 提频方案

提频方案的本质还是利用mtk的power机制

在日常使用过程中，有些重点的场景，经常遇到卡顿的问题，特别是在负载的情况下，很容易出现资源不足造成卡顿，影响体验。我们为此构建了一套UX Boost方案，在识别到处于关键场景时，主动提升CPU频率，并将动效相关的线程摆到满足性能的CPU上运行，从而提升整体的使用体验。基于对场景的精准把控，此项提升并不影响功耗的表现。

vendor/transsion/THUB/thub_core_factory_core_services/lib/src/main/java/com/transsion/hubcore/cpubooster/feature/TranUXBoostMgr.java

package com.transsion.hubcore.cpubooster.feature;

import com.mediatek.powerhalmgr.PowerHalMgr;
import com.mediatek.powerhalmgr.PowerHalMgrFactory;
import android.util.SparseArray;

public class TranUXBoostMgr {

    private static final int MTKPOWER_HINT_LAUNCHER_ANIMA = 101;
    private static final int MTKPOWER_HINT_ANIMATING_BOOST = 103;
    private static final int MTKPOWER_HINT_VOLUME_CHANGE = 104;
    private static final int MTKPOWER_HINT_REMOTE_ANIMA = 109;
    private static TranUXBoostMgr sInstance;
    private static PowerHalMgr mPowerHalService;
    private final SparseArray<Integer> mPowerHandle = new SparseArray<>();

    private TranUXBoostMgr() {
        mPowerHalService = PowerHalMgrFactory.getInstance().makePowerHalMgr();
    }

    public static synchronized TranUXBoostMgr get() {
        if (sInstance == null) {
            sInstance = new TranUXBoostMgr();
        }
        return sInstance;
    }

    public void startSurfaceAnimationBoost(long maxDuration) {
        if (mPowerHalService != null) {
            mPowerHalService.perfCusLockHint(MTKPOWER_HINT_LAUNCHER_ANIMA, (int)maxDuration);
        }
    }

    public void startWindowAnimationBoost() {
        if (mPowerHalService != null) {
            releaseWindowAnimationBoost();
            int powerHandle = mPowerHalService.perfCusLockHint(MTKPOWER_HINT_ANIMATING_BOOST, 1000);
            mPowerHandle.put(MTKPOWER_HINT_ANIMATING_BOOST, powerHandle);
        }
    }

    public void releaseWindowAnimationBoost() {
        int powerHandle = mPowerHandle.get(MTKPOWER_HINT_ANIMATING_BOOST, 0);
        if (mPowerHalService != null && powerHandle != 0) {
            mPowerHalService.perfLockRelease(powerHandle);
        }
    }

    public void startTransitionBoost() {
        if (mPowerHalService != null) {
            releaseTransitionBoost();
            int powerHandle = mPowerHalService.perfCusLockHint(MTKPOWER_HINT_LAUNCHER_ANIMA, 1000);
            mPowerHandle.put(MTKPOWER_HINT_LAUNCHER_ANIMA, powerHandle);
        }
    }

    public void releaseTransitionBoost() {
        int powerHandle = mPowerHandle.get(MTKPOWER_HINT_LAUNCHER_ANIMA, 0);
        if (mPowerHalService != null && powerHandle != 0) {
            mPowerHalService.perfLockRelease(powerHandle);
        }
    }

    public void startAudioVolumeBoost() {
        if (mPowerHalService != null) {
            mPowerHalService.perfCusLockHint(MTKPOWER_HINT_VOLUME_CHANGE, 80);
        }
    }

    public void startRemoteAnimationBoost() {
        if (mPowerHalService != null) {
            int powerHandle = mPowerHalService.perfCusLockHint(MTKPOWER_HINT_REMOTE_ANIMA, 1000);
            mPowerHandle.put(MTKPOWER_HINT_REMOTE_ANIMA, powerHandle);
        }
    }

    public void releaseRemoteAnimationBoost() {
        int powerHandle = mPowerHandle.get(MTKPOWER_HINT_REMOTE_ANIMA, 0);
        if (mPowerHalService != null && powerHandle != 0) {
            mPowerHalService.perfLockRelease(powerHandle);
        }
    }

    public void startRecentsAnimationBoost() {
        if (mPowerHalService != null) {
            int powerHandle = mPowerHalService.perfCusLockHint(MTKPOWER_HINT_REMOTE_ANIMA, 1000);
            mPowerHandle.put(MTKPOWER_HINT_REMOTE_ANIMA, powerHandle);
        }
    }

    public void releaseRecentsAnimationBoost() {
        int powerHandle = mPowerHandle.get(MTKPOWER_HINT_REMOTE_ANIMA, 0);
        if (mPowerHalService != null && powerHandle != 0) {
            mPowerHalService.perfLockRelease(powerHandle);
        }
    }
}

                    mPowerHintOpts.startRemoteAnimationBoost();
                }
                break;
            default:
                Slog.e(TAG, "not support type, type=" + type);
        }
    }

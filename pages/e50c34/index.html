public void setRotationBoost(boolean enable) {
    int boostTime = 0;
    // boostTime > 0 means enable boost
    if (enable && !mIsRotationBoostEnable) {
        boostTime = PowerHalManager.ROTATE_BOOST_TIME;
    } else if (!enable && mIsRotationBoostEnable) {
        boostTime = 0;
    }

    //Log.e(TAG, "<setRotationBoost> do boost with " + enable);
    mPowerHalWrap.setRotationBoost(boostTime);
    mIsRotationBoostEnable = enable;
}

lib\src\main\java\com\transsion\hubcore\spdopts\cpubooster\feature\TranLauncherBoosterMgr.java
/**
    * boost start interface for Luancher
    * @param type boost type
    */
public void boostStartInLauncher(int type) {
    if (!isEnable()) {
        return;
    }

    if (DEBUG) {
        Slog.d(TAG, "boostStartInLauncher, type=" + type);
    }

    switch (type) {
        case SCENE_ZERO_SCREEN:
            mPowerHintOpts.startLauncherBoost();
            mTranCpuBoosterMgr.boostSceneStart(mTranCpuBoosterMgr.BOOST_SCENE_7);
            break;
        case SCENE_NORMAL:
            mPowerHintOpts.startLauncherBoost();
            break;
        case SCENE_SPECIAL:
            mPowerHintOpts.startLauncherBoost();
            if (ITranSurfaceControl.Instance().requiresClientComposition()){
                mTranCpuBoosterMgr.boostSceneStart(mTranCpuBoosterMgr.BOOST_SCENE_4);
            }
            break;
        case SCENE_UNLOCK:
            mPowerHintOpts.startTransitionBoost();
            mTranCpuBoosterMgr.boostUnlockStart(true);
            break;
        case SCENE_MAYBE_SWIPE_UP:
            mTranCpuBoosterMgr.boostMaybeSwipeUp();
            break;
        default:
            Slog.e(TAG, "not support type, type=" + type);
    }
}

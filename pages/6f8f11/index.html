# neo4j

package com.java.data;

import org.neo4j.driver.AuthTokens;
import org.neo4j.driver.Driver;
import org.neo4j.driver.GraphDatabase;
import org.neo4j.driver.Session;
import org.neo4j.driver.Transaction;
import org.neo4j.driver.exceptions.Neo4jException;

public class Neo4jConnectionExample {

    public static void main(String[] args) {
        // 连接到Neo4j数据库
        Driver driver = GraphDatabase.driver("bolt://121.40.221.224:8065", AuthTokens.basic("neo4j", "xxx"));

        try (Session session = driver.session()) {
            // 插入商品和用户信息
            String createProductsCypher = "CREATE (:P1 {productId: 1, productName: '智能手机', productDescription: '用于通信的智能手机', productQuantity: 100}),"
                    + "(:P1 {productId: 2, productName: '耳机', productDescription: '用于听音乐的耳机', productQuantity: 200}),"
                    + "(:P1 {productId: 3, productName: '键盘', productDescription: '用于输入的键盘', productQuantity: 300}),"
                    + "(:P1 {productId: 4, productName: '显示器', productDescription: '用于显示的显示器', productQuantity: 150}),"
                    + "(:P1 {productId: 5, productName: '苹果耳机', productDescription: '苹果品牌的耳机', productQuantity: 50}),"
                    + "(:P1 {productId: 6, productName: '小米手机', productDescription: '小米品牌的智能手机', productQuantity: 90})";
            String createUsersCypher = "CREATE (:U1 {userId: 1, userName: 'Alice'}),"
                    + "(:U1 {userId: 2, userName: 'Bob'}),"
                    + "(:U1 {userId: 3, userName: 'Charlie'})";

            // 创建用户与商品之间的关系
            String createRelationshipsCypher = "MATCH (u:U1 {userName: 'Alice'}), (p:P1 {productName: '智能手机'}) "
                    + "CREATE (u)-[:PURCHASED]->(p) "
                    + "WITH u, p "
                    + "MATCH (u:U1 {userName: 'Bob'}), (p1:P1 {productName: '键盘'}), (p2:P1 {productName: '显示器'}) "
                    + "CREATE (u)-[:PURCHASED]->(p1) "
                    + "CREATE (u)-[:PURCHASED]->(p2)";

            try (Transaction tx = session.beginTransaction()) {
                tx.run(createProductsCypher);
                tx.run(createUsersCypher);
                tx.run(createRelationshipsCypher);
                tx.commit();
                System.out.println("Data has been inserted successfully.");
            } catch (Neo4jException e) {
                System.err.println("Error inserting data into Neo4j: " + e.getMessage());
            }
        } catch (Exception e) {
            System.err.println("Error while connecting to Neo4j: " + e.getMessage());
        } finally {
            driver.close(); // 关闭连接
        }
    }
}

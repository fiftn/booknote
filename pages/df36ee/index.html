# 简单模块实践

#include <linux/module.h>
#include <linux/kernel.h>
#include <linux/init.h>

static int age=0;
static char* name=NULL;

module_param(age, int , S_IRUGO);
module_param(name, charp, S_IRUGO);

static int __init init_hello(void)
{
    printk(KERN_DEBUG "Hello %s, are you %d years old?\n", name, age);
    return 0; // 添加返回值
}

static void __exit exit_hello(void)
{
    printk(KERN_DEBUG "bye bye world\n");
}

module_init(init_hello);
module_exit(exit_hello);

MODULE_LICENSE("GPL");
MODULE_AUTHOR("xiaojiang"); // 修正拼写错误
MODULE_VERSION("V0.11");
MODULE_DESCRIPTION("HELLO WORLD!");

# 内核源码所在位置 or 至少是你当前内核版本的头文件
KERNEL_DIR  := /lib/modules/$(shell uname -r)/build
# 当前目录
CURRENR_DIR	:= $(shell pwd)

# obj-m 表示把文件作为"模块"编译，不会编译进内核，但会生成一个独立的 ko 文件
# obj-y 则是直接编译进内核，即加入到内核源码中
# 编译产物是hello.0，会寻找hello.c
obj-m	:= hello.o 

# 这行定义了一个名为 build 的目标，并将其依赖项设为 kernel_modules
build: kernel_modules

# $(MAKE)：调用 make 命令。
# -C $(KERNEL_DIR)：切换到内核源码目录。
# M=$(CURRENR_DIR)：传递当前工作目录给内核构建系统。
# modules：这是在内核构建目录中执行的目标，表示构建模块。
kernel_modules:
	$(MAKE) -C $(KERNEL_DIR) M=$(CURRENR_DIR) modules

# clean一下
.PHONY:clean
clean:
	$(MAKE) -C $(KERNEL_DIR) M=$(CURRENR_DIR) clean

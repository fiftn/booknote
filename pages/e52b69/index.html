# CPU相关命令

//默认值是 -1 -1 
adb shell "echo 0 0 > /proc/ppm/policy/ut_fix_freq_idx" //给cpu的频率设置最大

//性能模式
//默认值是sugov_ext
cd sys/devices/system/cpu/cpufreq
echo performance > ./policy0/scaling_governor
echo performance > ./policy4/scaling_governor
echo performance > ./policy7/scaling_governor

//设置cpu簇 频率的上下限
//cpu簇概念：0，4，7   分别表示0-3cpu核 4-6cpu核 7cpu核
adb shell  //进入adb环境
echo "4 1200000 2150000" > /proc/cpudvfs/cpufreq_debug//设置cpu簇4的上下限,最小频率不低于12000
//设置完后进行查看：
cd /sys/devices/system/cpu/cpufreq/policy4
cat cpuinfo_min_freq
cat cpuinfo_max_freq
cat cpuinfo_cur_freq//这个是当前的cpu的值

D1300（平台）:
echo "1 1600000" > /proc/ppm/policy/hard_userlimit_min_cpu_freq//设置中核组最小值为1.6
echo "1 1600000" > /proc/ppm/policy/hard_userlimit_max_cpu_freq

//查看cpu挡位
Kernel 4.14/4.19
cat /proc/cpufreq/MT_CPU_DVFS_LL/cpufreq_oppidx
Kernel 5.10/5.15
cat /sys/devices/system/cpu/cpufreq/policy*/scaling_available_frequencies

//查看当前cpu频率
Kernel 4.14/4.19
cat /proc/cpufreq/MT_CPU_DVFS_L/cpufreq_freq
cat /sys/devices/system/cpu/cpu0/cpufreq/scaling_cur_freq
Kernel 5.10/5.15
cat /sys/devices/system/cpu/cpuX/cpufreq/cpuinfo_cur_freq
    
//查看cpu核的所有频率挡位
cat  /sys/devices/system/cpu/cpufreq/policy0/scaling_available_frequencies 
 
//cpu关核
Kernel 4.14/4.19/5.10/5.15
adb shell "echo 0 > /proc/hps/enabled" 
adb shell "echo 0 > /sys/devices/system/cpu/cpuX/online" 关掉
adb shell "echo 1 > /sys/devices/system/cpu/cpu1/online" 打开

# GPU相关命令

Kernel 4.14/4.19 :
//查看当前频率
adb shell "cat /proc/gpufreq/gpufreq_var_dump"
//拿到gpu挡位
adb shell "cat /proc/gpufreq/gpufreq_opp_dump"
//输入最大挡位进去
adb shell "echo ***000 > /proc/gpufreq/gpufreq_opp_freq"
//查看当前挡位，检查是否提频
adb shell "cat /proc/gpufreq/gpufreq_var_dump"

//定GPU天花板OPP档位
adb shell "echo 1 > /d/ged/hal/custom_upbound_gpu_freq"
//定频
adb shell "echo FREQ  > proc/gpufreq/gpufreq_opp_freq"

Kernel 5.10/5.15 :
//查看当前频率

//查看当前gpu挡位
adb shell "cat /proc/gpufreqv2/gpu_working_opp_table"
adb shell "cat /proc/gpufreqv2/stack_working_opp_table"
(有stack看stack，没stack看gpu)

压缩python文件：

#!/usr/bin/env python3

import os
import re
import subprocess
import sys
import base64

def encode(infile, outfile):
    fd = open(infile)
    try:
        buf = fd.read()
    finally:
        fd.close()

    b = bytearray(base64.b64encode(buf.encode('utf-8')))
    for i in range(len(b)):
        b[i] += 33

    open(outfile, 'wb').write(b)


def decode(infile, outfile):
    b = bytearray(open(infile, 'rb').read())
    for i in range(len(b)):
        b[i] -= 33

    d = bytes.decode(base64.b64decode(b))
    open(outfile, 'w').write(d)


def usage():
    print("Usage: " + sys.argv[0] + " -e/d <input file> <output file>")

if __name__ == '__main__' :
    argc = len(sys.argv)
    if argc != 4:
        usage()
        exit()

    if(sys.argv[1] == "-e"):
        encode(sys.argv[2], sys.argv[3])
    elif(sys.argv[1] == "-d"):
        decode(sys.argv[2], sys.argv[3])
    else:
        usage()

# adb remount

  1. 打开设置——关于手机——多次点击版本号，打开开发者模式；返回上一级——进入系统——高级——开发者选项——打开开发者选项中的OEM解锁开关
  2. adb reboot bootloader（进入fastboot模式）
  3. fastboot devices（查看设备）
  4. fastboot flashing unlock（解锁）
  5. 按照屏幕提示，按物理音量上键，进行解锁（没提示直接按即可）
  6. fastboot reboot

到此解锁成功，adb remount后会提示你重启手机，重启后即解锁成功

更好的方式：bat脚本命令一键执行：

@echo off

echo "!!!!手机上打开 oem unlock: setting -> system -> Developer options -> OEM unlocking!!!!"

timeout 30
adb reboot bootloader
echo "starting fastboot, plz wait!"

fastboot oem tran_skip_confirm_key
fastboot flashing unlock
echo "processing 1/6, just a sec..."
fastboot reboot

echo "processing 2/6, just a sec..."
adb wait-for-device
adb root

echo "processing 3/6, just a sec..."
adb wait-for-device
adb disable-verity

echo "processing 4/6, just a sec..."
adb wait-for-device
adb reboot

echo "processing 5/6, just a sec..."
adb wait-for-device
adb root

echo "processing 6/6, just a sec..."
adb wait-for-device
adb remount

echo "!!!!--------------------remount done~---------------------------!!!!"

timeout 30


adb shell pm disable com.transsion.overlaysuw
adb shell settings put global device_provisioned 1
adb shell settings put secure user_setup_complete 1

timeout 30

# 温升限频

# kernel4.19分析方法

  • 观察trace，分析频率，以及卡顿的原因是否由running 和 sleeping耗时长导致

频率很低

主线程ui抢不到资源，并且长时间处于sleeping和running状态下

  • 分析日志，从cpu温度、限制频率关键字、系统负载情况三方面进一步分析确认

在kernel log重搜索关键字ppm，即可看到限频信息，关于ppm日志，请看mtk文档学习这一栏

继续搜索关键字power_limit，即可确定cpu温度信息，可以看到此时cpu温度达到60多

继续搜索关键字TRAN PERF，即可确认系统这时的loading状态，该关键字为公司特加的

关闭温升，辅助验证方法： 关闭温控步骤如下：

1) root 版本remount解锁后进入/vendor/etc/.tp目录:
   adb root
   adb remount
   adb shell
   cd /vendor/etc/.tp
2) 备份thermal.conf:
   cp -af thermal.conf thermal.conf.bak
3) 使用.ht120.mtc：
   cp -af .ht120.mtc thermal.conf
4) 重启系统
   或者
   adb shell "thermal_manager /vendor/etc/.tp/.ht120.mtc"
   adb shell "cat /data/vendor/.tp/.settings"
   结果为 .ht120.mtc，则disable thermal成功

注意：此方法不一定会真的关闭，关闭后如有问题 重新看kernel log.

# kernel5.10分析方法

  • 分析trace，5.10上有新的方法更好确认

需要使用MTK的工具perfconv_eve_v2.27转换才可以看到

  • main log中搜索关键字throttlingStatus

第一个红框——CPU/GPU/NPU：被限制的对下，cpu表示当前限制了cpu

第二个红框——SERVRE：发起限制的对下，这里表示是server在主动直线频率，如果这里是NONE，则表示无限制

第三个红框——61.471：表示限制到某个温度

其中的第一行的json示意：

CPU核心数为8，即系统中有8个CPU核心。
每个CPU核心的负载情况在"per"字段中列出，以百分比表示。这里显示的数值分别对应8个CPU核心的负载情况。
"info"字段中记录了一些关于CPU的其他信息，包括时间片（timeslice）的数量以及其他统计数据。
"top"字段列出了使用CPU最多的一些进程和它们的CPU占用百分比。

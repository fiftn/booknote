package com.transsion.hubcore.spdopts.ioprefetch.implement;
import android.content.Intent;
import android.content.pm.PackageManagerInternal;
import com.android.server.wm.ActivityRecord;
import android.annotation.NonNull;
import java.io.PrintWriter;
import android.util.Slog;

import com.transsion.hubcore.spdopts.ioprefetch.feature.TranIOPrefetch;
import com.transsion.hubcore.ioprefetch.ITranIOPrefetchComponent;
public class TranIOPrefetchComponentImpl implements ITranIOPrefetchComponent {

    @Override
    public void startLauncherAction(PackageManagerInternal pm, Intent intent, int action) {
        TranIOPrefetch.get().startLauncherAction(pm, intent, action);
    }

    @Override
    public void hookOnFirstWindowDrawn(ActivityRecord record, PackageManagerInternal pm, String windowTag) {
        TranIOPrefetch.get().hookOnFirstWindowDrawn(record, pm, windowTag);
    }

    @Override
    public void startActivityHook( @NonNull PackageManagerInternal pm,
                                  @NonNull Intent intent, @NonNull boolean processRunning, @NonNull boolean restrictedBgActivity) {
        TranIOPrefetch.get().startActivityHook(pm, intent, processRunning, restrictedBgActivity);
    }

    @Override
    public void startActivityLaunchHook(@NonNull PackageManagerInternal pm,
                                        @NonNull Intent intent, @NonNull boolean restrictedBgActivity) {
        TranIOPrefetch.get().startActivityLaunchHook(pm, intent, restrictedBgActivity);
    }

    @Override
    public void hookStartProcess(String processName) {
        TranIOPrefetch.get().hookStartProcess(processName);
    }

    @Override
    public boolean doDump(String cmd, PrintWriter pw, String[] args, int opti) {
        if("perio".equals(cmd)) {
            if(opti >= args.length) {
                pw.println("perio_config args failed!");
            } else {
                String value = args[opti];
                if("on".equals(value)) {
                    TranIOPrefetch.get().doDumpPerIOSwitch(true);
                } else if("off".equals(value)) {
                    TranIOPrefetch.get().doDumpPerIOSwitch(false);
                } else
                    pw.println("perio arg is on or off!");
            }
            return true;
        }
        return false;
    }
}

package com.transsion.hubcore.spdopts.ioprefetch.feature;

import android.os.SystemProperties;
import android.util.Slog;
import android.content.Intent;
import android.os.Handler;
import android.os.HandlerThread;
import android.os.Message;
import android.os.Process;
import android.content.pm.PackageManagerInternal;
import com.android.server.pm.pkg.AndroidPackage;
import android.content.res.ApkAssets;
import com.android.server.wm.ActivityRecord;
import java.io.File;
import android.annotation.Nullable;
import android.annotation.NonNull;
import java.io.IOException;
import java.io.IOException;
import android.os.Trace;
import java.io.FileDescriptor;
import android.system.ErrnoException;
import com.transsion.hubsdk.trancare.cloudengine.ITranCloudEngineCallback;
import com.transsion.hubsdk.trancare.trancare.TranTrancareManager;
import android.content.res.ApkAssets;
import android.system.Os;
import android.system.OsConstants;
import com.transsion.hubcore.server.preio.TranPreIONative;;


public class TranIOPrefetch {
    private static final String TAG = "TranIOPreFetch";
    private static final boolean SUPPORT = SystemProperties.getInt("ro.transsion.perio.support", 0) > 0;
    private static final boolean DEBUG = SystemProperties.getInt("persist.sys.tran.io.prefetch.debug", 0) > 0;
    private static final String CLOUD_KEY = "1001000068";
    private static final String CLOUD_VERSION = "v1.x";
    private static boolean mCloudEnable = true;
    private static boolean mPerIOEnable = true;
    //如果打开inner阶段预加载会导致与启动线程抢占IO的情况，默认关闭
    private static boolean mInnerEnable = false;
    public static final int START_ACTION_DOWN = 1;
    public static final int START_ACTION_PRESS  = 2;
    public static final int START_ACTION_SCROLL = 3;
    public static final int START_ACTIVITY_INNER = 4;
    public static final int START_WINDOW_DRAWN = 5;
    private volatile int lastReadPosition = -1;
    private static final int kIdealIoTransferSizeBytes = 128 * 1024;//128k
    private String mPackageName;
    private volatile boolean needInterrupt = false;
    private String mApkPath;
    private String mVdexPath;
    private String mOdexPath;
    private boolean mVdexDone = false;
    private boolean mOdexDone = false;
    private static final int APK_FILE = 1;
    private static final int VDEX_FILE = 2;
    private static final int ODEX_FILE = 3;
    private volatile static TranIOPrefetch sInstance;
    private HandlerThread mHandlerThread;
    private Handler mHandler;

    private TranIOPrefetch() {
        CloudCallback callback = new CloudCallback();
        TranTrancareManager.regCloudEngineCallback(CLOUD_KEY, CLOUD_VERSION, callback);
        mHandlerThread = new HandlerThread("TranPerIOThread");
        mHandlerThread.start();
        mHandler = new Handler(mHandlerThread.getLooper(), new MyHandlerCallback());
        Process.setThreadPriority(mHandlerThread.getThreadId(), Process.THREAD_PRIORITY_BACKGROUND);
    }

    public static TranIOPrefetch get() {
        if (sInstance == null) {
            synchronized (TranIOPrefetch.class){
                if(sInstance == null) {
                    sInstance = new TranIOPrefetch();
                }
            }
        }
        return sInstance;
    }

    private class CloudCallback extends ITranCloudEngineCallback.Stub {
        @Override
        public void onUpdate(String key, boolean filestate, String config) {
            if (CLOUD_KEY.equals(key)) {
                if (DEBUG) {
                    Slog.d(TAG, "[onUpdate] key = " + key + " filestate = " + filestate + " config = " + config);
                }
                mCloudEnable = TranTrancareManager.getEnabled(key, false);
                TranTrancareManager.feedBack(key, true);
            }
        }
    }

    private class MyHandlerCallback implements Handler.Callback {
        @Override
        public boolean handleMessage(Message msg) {
            if (DEBUG) {
                Slog.d(TAG, "MyHandlerCallback handleMessage, msg=" + msg.what);
            }

            AndroidPackage androidPackage = null;

            switch (msg.what) {
                case START_ACTION_DOWN:
                    androidPackage = (AndroidPackage) msg.obj;
                    needInterrupt = false;
                    captureFile(androidPackage);
                    prefetchFile(START_ACTION_DOWN);
                    break;
                case START_ACTIVITY_INNER:
                    //start activity
                    if(mInnerEnable) {
                        androidPackage = (AndroidPackage) msg.obj;
                        needInterrupt = false;
                        if (mVdexPath == null || mOdexPath == null) {
                            captureFile(androidPackage);
                        }
                        prefetchFile(START_ACTIVITY_INNER);
                    }
                    break;
                case START_WINDOW_DRAWN:
                    //action after lunch
                    lastReadPosition = -1;
                    mApkPath = null;
                    mVdexPath = null;
                    mOdexPath = null;
                    mVdexDone = false;
                    mOdexDone = false;
                    break;
                default:
                    Slog.e(TAG, "not support msg=" + msg.what);
            }
            return true;
        }
    }

    private void captureFile(AndroidPackage androidPackage) {
        if(androidPackage == null) {
            return;
        }

        mApkPath = androidPackage.getBaseApkPath();
        if(mApkPath == null) {
            return;
        }

        String dexPath = mApkPath.substring(0, mApkPath.lastIndexOf("/"));
        String dexPath32 = dexPath + "/oat/arm/";
        String dexPath64 = dexPath + "/oat/arm64/";

        try{
            File dexDir32 = new File(dexPath32);
            File dexDir64 = new File(dexPath64);
            if(dexDir32.exists() && dexDir32.isDirectory()) {
                for (File file : dexDir32.listFiles()) {
                    if (file.getName().endsWith(".vdex")) {
                        mVdexPath = file.getAbsolutePath();
                    } else if (file.getName().endsWith(".odex")) {
                        mOdexPath = file.getAbsolutePath();
                    }
                }
            } else if (dexDir64.exists() && dexDir64.isDirectory()) {
                for (File file : dexDir64.listFiles()) {
                    if (file.getName().endsWith(".vdex")) {
                        mVdexPath = file.getAbsolutePath();
                    } else if (file.getName().endsWith(".odex")) {
                        mOdexPath = file.getAbsolutePath();
                    }
                }
            }
        } catch (Exception e) {
            Slog.d(TAG, "prefetchSystemAPPFile is fail!");
        }
    }

    private void prefetchFile(int type) {
        switch (type) {
            case START_ACTION_DOWN:
                prefetch(APK_FILE, mApkPath);
                mVdexDone = prefetch(VDEX_FILE, mVdexPath);
                if(mVdexDone) {
                    mOdexDone = prefetch(ODEX_FILE, mOdexPath);
                }
                break;
            case START_ACTIVITY_INNER:
                if(!mVdexDone) {
                    prefetch(VDEX_FILE, mVdexPath);
                } else if(!mOdexDone) {
                    prefetch(ODEX_FILE, mOdexPath);
                }
                break;
            default:
                break;
        }
    }

    private boolean prefetch(int fileType, String path) {
        if(path == null) {
            return false;
        }

        boolean readIsDone = true;

        switch (fileType) {
            case APK_FILE:
                try {
                     ApkAssets.loadFromPath(path);
                } catch (IOException e) {
                    Slog.e(TAG, "loadFromPath is fail!");
                }
                break;
            case VDEX_FILE:
            case ODEX_FILE:
                readIsDone = doReadahead(path);
                break;
            default:
                break;
        }

        return readIsDone;
    }

    private boolean doReadahead(String path) {
        if (path == null) {
            return false;
        }

        FileDescriptor fd = null;
        int readSize = 0;
        int start = 0;

        try {
            int openFlags = (OsConstants.O_RDONLY | OsConstants.O_CLOEXEC);
            fd = Os.open(path, openFlags, 0);
            readSize = (int) Math.min(Os.fstat(fd).st_size, Integer.MAX_VALUE);

            start = (lastReadPosition >= 0) ? lastReadPosition : 0;
            Trace.traceBegin(Trace.TRACE_TAG_VIEW, "IOPre_prefetch path = " + path + "start = " + start);

            for(;start < readSize;start += kIdealIoTransferSizeBytes) {
                if(needInterrupt) {
                    needInterrupt = false;
                    lastReadPosition = start;
                    return false;
                }
                int len = (int)Math.min(kIdealIoTransferSizeBytes, readSize);
                TranPreIONative.nativeReadahead(start, len, fd);
            }
            lastReadPosition = -1;
        } catch (ErrnoException ex) {
            Slog.e(TAG, "Could not open file " + path, ex);
        }finally {
            safeClose(fd);
            Trace.traceEnd(Trace.TRACE_TAG_VIEW);
        }

        return true;
    }
    //T-HUB Core[SPD]:added for preIO by fan.feng1 20230830 start

    /**
     * Close FD, swallowing irrelevant errors.
     */
    private static void safeClose(@Nullable FileDescriptor fd) {
        if (fd != null && fd.valid()) {
            try {
                Os.close(fd);
            } catch (ErrnoException ex) {
                // Swallow the exception: non-EBADF errors in close(2)
                // indicate deferred paging write errors, which we
                // don't care about here. The underlying file
                // descriptor is always closed.
                if (ex.errno == OsConstants.EBADF) {
                    throw new AssertionError(ex);
                }
            }
        }
    }

    /**
     * dumpsys activity animboost on/off
     *
     * @param enable on/off
     */
    public void doDumpPerIOSwitch(boolean enable) {
        mPerIOEnable = enable;
    }
    //T-HUB Core[SPD]:added for preIO by fan.feng1 20230830 end

    private void handlePreio(PackageManagerInternal pm, Intent intent, String pkg, int pid, int what) {
        if (mHandler == null) {
            Slog.e(TAG, "handlePreio error: handler is not inited");
            return;
        }

        mPackageName = pkg;
        if (mPackageName == null && intent != null) {
            mPackageName = intent.getComponent().getPackageName();
        }

        AndroidPackage androidPackage = null;

        if (mPackageName != null) {
            androidPackage = pm.getPackage(mPackageName);
        }
        if (androidPackage == null) {
            Slog.e(TAG, "Cannot find package for package: " + mPackageName);
            return;
        }
        try {
            Message msg = new Message();
            msg.obj = androidPackage;
            msg.arg1 = pid;
            msg.what = what;
            mHandler.sendMessage(msg);
        } catch (Exception e) {
            Slog.e(TAG, "handlePerio error:" + e.getMessage());
        }
    }

    private boolean isEnable() {
        return SUPPORT && mCloudEnable && mPerIOEnable;
    }

    /**
     * launcher事件触发:按下、滑动、长按
     *
     * @param pm
     * @param intent
     * @param action
     */
    public void startLauncherAction(PackageManagerInternal pm, Intent intent, int action) {
        if (!isEnable() || pm == null || intent == null) {
            Slog.e(TAG, "startLauncherAction is not enable");
            return;
        }

        String packageName = intent.getStringExtra("package");
        AndroidPackage androidPackage = null;

        if (packageName != null) {
            androidPackage = pm.getPackage(packageName);
        }
        if (androidPackage == null) {
            Slog.e(TAG, "Cannot find package for package: " + packageName);
            return;
        }

        if(action == START_ACTION_DOWN) {
            handlePreio(pm, intent, packageName, 0, START_ACTION_DOWN);
        } else {
            needInterrupt = true;
        }
    }

    /**
     * 应用启动完成后对apk内存数据进行收集
     *
     * @param record
     * @param pm
     * @param windowTag
     */
    public void hookOnFirstWindowDrawn(ActivityRecord record, PackageManagerInternal pm, String windowTag) {
        if (!isEnable() || record == null || pm == null || windowTag == null) {
            return;
        }

        if (DEBUG) {
            if (windowTag.contains("Splash Screen")) {
                Slog.d(TAG, "is app_start_time hookOnFirstWindowDrawn = ");
            }
        }

        if (windowTag.contains("Splash Screen") && mHandler != null) {
            String packageName = record.packageName;
            int pid = record.app.getPid();
            handlePreio(pm, null, packageName, pid, START_WINDOW_DRAWN);
        }
    }

    /**
     * 应用启动getResources hook
     *
     * @param processName
     */
    public void startActivityLaunchHook(@NonNull PackageManagerInternal pm,
                                        @NonNull Intent intent, @NonNull boolean restrictedBgActivity) {
        if (!isEnable() || restrictedBgActivity || needInterrupt) {
            return;
        }

        if (pm == null || intent == null || intent.getComponent() == null) {
            return;
        }

        final String packageName = intent.getComponent().getPackageName();
        AndroidPackage androidPackage = null;

        if (packageName != null) {
            androidPackage = pm.getPackage(packageName);
        }
        if (androidPackage == null) {
            Slog.e(TAG, "Cannot find package for package: " + packageName);
            return;
        }

        if (packageName.equals(mPackageName)) {
            needInterrupt = true;
        }
    }

    /**
     * 启动时进行预加载
     *
     * @param processName
     */
    public void startActivityHook(@NonNull PackageManagerInternal pm,
                                  @NonNull Intent intent, @NonNull boolean processRunning, @NonNull boolean restrictedBgActivity) {
        if (!isEnable() || processRunning || restrictedBgActivity) {
            return;
        }

        if (pm == null || intent == null || intent.getComponent() == null) {
            return;
        }

        final String packageName = intent.getComponent().getPackageName();
        AndroidPackage androidPackage = null;

        if (packageName != null) {
            androidPackage = pm.getPackage(packageName);
        }
        if (androidPackage == null) {
            Slog.e(TAG, "Cannot find package for package: " + packageName);
            return;
        }

        handlePreio(pm, intent, null, 0, START_ACTIVITY_INNER);
    }

    /**
     * 应用启动时，停止预加载
     *
     * @param processName
     */
    public void hookStartProcess(String processName) {
        if (!isEnable() || processName == null) {
            return;
        }
        if (processName.equals(mPackageName)) {
            needInterrupt = true;
        }
    }
}
